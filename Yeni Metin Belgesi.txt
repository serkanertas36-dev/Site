  <script>
    function handleImgError(img, baseName, altName) {
      const step = parseInt(img.dataset.retry || "0");
      const candidates = [
        baseName + ".jpg.jpg", baseName + ".jpeg", baseName + ".png", baseName.toUpperCase() + ".JPG", baseName + ".jgp"
      ];
      if (altName) {
        candidates.push(altName + ".jpg", altName + ".jpg.jpg", altName + ".png", altName + ".jpeg", altName + ".jgp");
      }
      if (step < candidates.length) {
        img.dataset.retry = step + 1;
        img.src = candidates[step];
      }
    }

        return { x: side.posX + localX * c + localZ * s, z: side.posZ - localX * s + localZ * c };
    }

    function createFramedGlass(pw, ph, fThick = 0.04) {
      const grp = new THREE.Group(); const glass = new THREE.Mesh(new THREE.BoxGeometry(pw, ph, 0.01), glassMat); grp.add(glass);
      const tbGeo = new THREE.BoxGeometry(pw, fThick, 0.03);
      const topF = new THREE.Mesh(tbGeo, frameMat); topF.position.y = ph / 2 - fThick / 2; const botF = new THREE.Mesh(tbGeo, frameMat); botF.position.y = -ph / 2 + fThick / 2;
      const lrGeo = new THREE.BoxGeometry(fThick, ph - fThick * 2, 0.03);
      const leftF = new THREE.Mesh(lrGeo, frameMat); leftF.position.x = -pw / 2 + fThick / 2; const rightF = new THREE.Mesh(lrGeo, frameMat); rightF.position.x = pw / 2 - fThick / 2;
      grp.add(topF, botF, leftF, rightF); return grp;
    }

    function createFramelessGlass(pw, ph) {
      const grp = new THREE.Group(); const glass = new THREE.Mesh(new THREE.BoxGeometry(pw - 0.005, ph, 0.01), glassMat); grp.add(glass);
      const tbGeo = new THREE.BoxGeometry(pw, 0.04, 0.03);
      const topF = new THREE.Mesh(tbGeo, frameMat); topF.position.y = ph / 2 - 0.02; const botF = new THREE.Mesh(tbGeo, frameMat); botF.position.y = -ph / 2 + 0.02;
      grp.add(topF, botF); return grp;
    }

    function createHingedAluminumDoor(pw, ph, openDirection = 'left') {
      const pivotGroup = new THREE.Group(); const doorPanel = new THREE.Group(); const fThick = 0.06;
      const glass = new THREE.Mesh(new THREE.BoxGeometry(pw - fThick*2, ph - fThick*2, 0.01), glassMat); glass.position.z = 0.005; doorPanel.add(glass);
      const tbGeo = new THREE.BoxGeometry(pw, fThick, 0.04);
      const topF = new THREE.Mesh(tbGeo, frameMat); topF.position.y = ph/2 - fThick/2; const botF = new THREE.Mesh(tbGeo, frameMat); botF.position.y = -ph/2 + fThick/2;
      const lrGeo = new THREE.BoxGeometry(fThick, ph, 0.04);
      const leftF = new THREE.Mesh(lrGeo, frameMat); leftF.position.x = -pw/2 + fThick/2; const rightF = new THREE.Mesh(lrGeo, frameMat); rightF.position.x = pw/2 - fThick/2;

    function buildSystem() {
      roofActionElements = []; glassPanels = []; zipPanels = [];
      while (systemGroup.children.length > 0) systemGroup.remove(systemGroup.children[0]);

      const W = parseFloat(document.getElementById('width').value) || 5.0; const D = parseFloat(document.getElementById('depth').value) || 3.5;
      const FH = parseFloat(document.getElementById('frontHeight').value) || 2.5; const BH = FH + 0.15;

      function createBox(w, h, d, x, y, z, rx = 0) {
        const mesh = new THREE.Mesh(new THREE.BoxGeometry(w, h, d), frameMat); mesh.position.set(x, y, z); mesh.rotation.x = rx; mesh.castShadow = true; systemGroup.add(mesh); return mesh;
      }

      createBox(POST_SIZE, FH, POST_SIZE, -W / 2 + POST_SIZE / 2, FH / 2, D / 2 - POST_SIZE / 2); createBox(POST_SIZE, FH, POST_SIZE,  W / 2 - POST_SIZE / 2, FH / 2, D / 2 - POST_SIZE / 2);
      createBox(POST_SIZE, BH, POST_SIZE, -W / 2 + POST_SIZE / 2, BH / 2, -D / 2 + POST_SIZE / 2); createBox(POST_SIZE, BH, POST_SIZE,  W / 2 - POST_SIZE / 2, BH / 2, -D / 2 + POST_SIZE / 2);
      createBox(W, BEAM_HEIGHT, BEAM_WIDTH, 0, FH - BEAM_HEIGHT / 2, D / 2 - BEAM_WIDTH / 2); createBox(W, BEAM_HEIGHT, BEAM_WIDTH, 0, BH - BEAM_HEIGHT / 2, -D / 2 + BEAM_WIDTH / 2);

      const slopeLength = Math.sqrt(Math.pow(D, 2) + Math.pow(BH - FH, 2)); const pitchAngle = Math.atan2(BH - FH, D); const roofCenterY = (FH + BH) / 2 - BEAM_HEIGHT / 2 + 0.08;
      createBox(BEAM_WIDTH, BEAM_HEIGHT, slopeLength, -W / 2 + BEAM_WIDTH / 2, roofCenterY, 0, pitchAngle); createBox(BEAM_WIDTH, BEAM_HEIGHT, slopeLength,  W / 2 - BEAM_WIDTH / 2, roofCenterY, 0, pitchAngle);

      if (currentRoof === 'bioclimatic' || currentRoof === 'rolling_roof') {
        const count = Math.ceil(D / 0.22); const step = D / count;
        for (let i = 0; i < count; i++) {
          const louver = new THREE.Mesh(new THREE.BoxGeometry(W - BEAM_WIDTH * 2, 0.03, step + 0.005), louverMat);
          louver.position.set(0, roofCenterY, -D / 2 + step * (i + 0.5)); louver.userData = { baseZ: louver.position.z, index: i }; louver.castShadow = true;
          systemGroup.add(louver); roofActionElements.push(louver);
        }
      } else if (currentRoof === 'pergola') {
        const segments = 12; const segStep = D / segments;
        for (let i = 0; i < segments; i++) {
          const fGroup = new THREE.Group(); const fabric = new THREE.Mesh(new THREE.BoxGeometry(W - 0.15, 0.02, segStep * 0.95), new THREE.MeshStandardMaterial({ color: 0xf8fafc })); fGroup.add(fabric);
          fGroup.position.set(0, roofCenterY, -D / 2 + segStep * (i + 0.5)); fGroup.userData = { baseZ: fGroup.position.z, index: i }; systemGroup.add(fGroup); roofActionElements.push(fGroup);
        }
      } else if (currentRoof === 'veranda') {
        const rG = new THREE.Mesh(new THREE.BoxGeometry(W - 0.05, 0.02, slopeLength), glassMat); rG.position.set(0, roofCenterY, 0); rG.rotation.x = pitchAngle; systemGroup.add(rG);
        const rafterCount = Math.ceil(W / 0.85); const rafterSpacing = W / rafterCount;
        for (let i = 1; i < rafterCount; i++) {
          const rafter = new THREE.Mesh(new THREE.BoxGeometry(0.06, 0.12, slopeLength + 0.1), frameMat); rafter.position.set(-W / 2 + i * rafterSpacing, roofCenterY - 0.02, 0); rafter.rotation.x = pitchAngle; systemGroup.add(rafter);
        }
      }

      const sides = [
        { id: 'Front', w: W - POST_SIZE * 2, h: FH - BEAM_HEIGHT, posX: 0, posZ: D / 2 - POST_SIZE / 2, rotY: 0, xDir: 0, zDir: 1 },
        { id: 'Back',  w: W - POST_SIZE * 2, h: BH - BEAM_HEIGHT, posX: 0, posZ: -D / 2 + POST_SIZE / 2, rotY: Math.PI, xDir: 0, zDir: -1 },
        { id: 'Left',  w: D - POST_SIZE * 2, h: (FH + BH) / 2 - BEAM_HEIGHT, posX: -W / 2 + POST_SIZE / 2, posZ: 0, rotY: -Math.PI / 2, xDir: -1, zDir: 0 },
        { id: 'Right', w: D - POST_SIZE * 2, h: (FH + BH) / 2 - BEAM_HEIGHT, posX: W / 2 - POST_SIZE / 2, posZ: 0, rotY: Math.PI / 2, xDir: 1, zDir: 0 }
      ];

      let hasGlass = false; let hasZip = false; let closedSides = 0; let hasInsulatedSystem = false;

      sides.forEach(side => {
        const type = document.getElementById('glazing' + side.id).value;
        if (type !== 'none') closedSides++;

        const isDoorCombined = type.includes('door_'); const isLeft = type.includes('_left');
        const isGuillotine = type.includes('guillotine'); const isSliding = type.includes('sliding'); const isCamBalkon = type.includes('cambalkon');
        
        let doorW = isDoorCombined ? 0.9 : 0; let mullionW = isDoorCombined ? 0.02 : 0; let spanW = side.w - doorW - mullionW;
        let doorCenterLocalX = 0; let mullionLocalX = 0; let glassCenterLocalX = 0;

        if (isDoorCombined) {
            if (isLeft) { doorCenterLocalX = -side.w / 2 + doorW / 2; mullionLocalX = -side.w / 2 + doorW + mullionW / 2; glassCenterLocalX = mullionLocalX + mullionW / 2 + spanW / 2; } 
            else { glassCenterLocalX = -side.w / 2 + spanW / 2; mullionLocalX = -side.w / 2 + spanW + mullionW / 2; doorCenterLocalX = side.w / 2 - doorW / 2; }
        } else { glassCenterLocalX = 0; spanW = side.w; }

        if (isDoorCombined) {
            const mullion = new THREE.Mesh(new THREE.BoxGeometry(mullionW, side.h, 0.12), frameMat);
            let mPos = getWPos(side, mullionLocalX, 0); mullion.position.set(mPos.x, side.h / 2, mPos.z); mullion.rotation.y = side.rotY; systemGroup.add(mullion);
        }

        if (isSliding || type === 'rts116') {
          const track = new THREE.Mesh(new THREE.BoxGeometry(spanW, 0.015, 0.08), frameMat);
          let tPos = getWPos(side, glassCenterLocalX, 0); track.position.set(tPos.x, 0.008, tPos.z); track.rotation.y = side.rotY; systemGroup.add(track);
        }

        if (type === 'wall') {
          const wall = new THREE.Mesh(new THREE.BoxGeometry(side.w, side.h, 0.08), wallMat);
          wall.position.set(side.posX, side.h / 2, side.posZ); wall.rotation.y = side.rotY; systemGroup.add(wall);
        } else if (type !== 'none') {
          hasGlass = true;
          if (isGuillotine) {
            for (let i = 0; i < 3; i++) {
                const pGroup = new THREE.Group(); let ph = side.h / 3; pGroup.add(createFramedGlass(spanW, ph, 0.04)); pGroup.rotation.y = side.rotY;
                let wPos = getWPos(side, glassCenterLocalX, (1 - i) * 0.04);
                pGroup.userData = { type: 'guillotine', isFixed: (i === 0), closedY: ph / 2 + i * ph, openY: ph / 2 + i * 0.05, closedX: wPos.x, closedZ: wPos.z };
                pGroup.position.set(wPos.x, pGroup.userData.closedY, wPos.z); systemGroup.add(pGroup); glassPanels.push(pGroup);
            }
          } else if (isSliding || type === 'rts116') {
            let panelCount = (type === 'rts116') ? 2 : (isDoorCombined ? 2 : 3); let overlap = (type === 'rts116') ? 0.08 : 0.06; let pw = (spanW + overlap * (panelCount - 1)) / panelCount;
            if(type === 'rts116' || type === 'rwt64') hasInsulatedSystem = true;
            for (let i = 0; i < panelCount; i++) {
                const pGroup = new THREE.Group(); pGroup.add(createFramedGlass(pw, side.h, 0.04)); pGroup.rotation.y = side.rotY;
                let localZ = (1 - i) * (type === 'rts116' ? 0.06 : 0.04);
                let closedPos = getWPos(side, (glassCenterLocalX - spanW / 2) + pw / 2 + i * (pw - overlap), localZ);
                let stackTargetX = isLeft ? (glassCenterLocalX + spanW/2 - pw/2) : (glassCenterLocalX - spanW/2 + pw/2);
                let openLocalX = stackTargetX + (isLeft ? -i*0.05 : i*0.05); if (!isDoorCombined) { openLocalX = (glassCenterLocalX - spanW/2 + pw/2) + i*0.05; }
                let openPos = getWPos(side, openLocalX, localZ);
                pGroup.position.set(closedPos.x, side.h / 2, closedPos.z); pGroup.userData = { type: 'sliding', isFixed: (i === 0), closedX: closedPos.x, closedZ: closedPos.z, openX: openPos.x, openZ: openPos.z };
                systemGroup.add(pGroup); glassPanels.push(pGroup);
            }
          } else if (type.includes('fd60') || isCamBalkon) {
            let panelCount = isCamBalkon ? Math.max(4, Math.ceil(spanW / 0.65)) : 4; let pw = spanW / panelCount;
            for (let i = 0; i < panelCount; i++) {
                const pGroup = new THREE.Group();
                if(isCamBalkon) pGroup.add(createFramelessGlass(pw, side.h)); else pGroup.add(createFramedGlass(pw, side.h, 0.04));
                let closedPos = getWPos(side, (glassCenterLocalX - spanW / 2) + pw / 2 + i * pw, 0);
                let openLocalX, openLocalZ, openRotY;
                if(isCamBalkon) {
                    openLocalX = (isLeft ? (glassCenterLocalX - spanW/2 + pw/2) : (glassCenterLocalX + spanW/2 - pw/2)) + (isLeft ? i*0.03 : -i*0.03); 
                    openLocalZ = 0.15; openRotY = side.rotY + (isLeft ? Math.PI/2.2 : -Math.PI/2.2);
                } else {
                    openLocalX = (glassCenterLocalX - spanW / 2) + (pw * 0.1) + i * pw * 0.15; openLocalZ = (i % 2 === 0) ? 0.3 : -0.3; openRotY = side.rotY + ((i % 2 === 0) ? 1.2 : -1.2);
                }
                let openPos = getWPos(side, openLocalX, openLocalZ);
                pGroup.userData = { type: isCamBalkon ? 'cambalkon' : 'fd60', closedX: closedPos.x, closedZ: closedPos.z, closedRotY: side.rotY, openX: openPos.x, openZ: openPos.z, openRotY };
                pGroup.position.set(closedPos.x, side.h / 2, closedPos.z); pGroup.rotation.y = side.rotY; systemGroup.add(pGroup); glassPanels.push(pGroup);
            }
          }

          if (isDoorCombined) {
              const doorDir = isLeft ? 'left' : 'right'; const doorGroup = createHingedAluminumDoor(doorW, side.h - 0.03, doorDir);
              let dClosedPos = getWPos(side, isLeft ? (-side.w / 2) : (side.w / 2), 0);
              doorGroup.position.set(dClosedPos.x, (side.h - 0.03) / 2, dClosedPos.z);
              doorGroup.userData = { type: 'hinged_door', closedAngle: side.rotY, openAngle: side.rotY + (isLeft ? -Math.PI / 2.2 : Math.PI / 2.2) };
              doorGroup.rotation.y = side.rotY; systemGroup.add(doorGroup); glassPanels.push(doorGroup);
          }
        }

        if (document.getElementById('zip' + side.id).checked) {
          hasZip = true; let zPos = getWPos(side, 0, 0.09);
          const cassette = new THREE.Mesh(new THREE.BoxGeometry(side.w + 0.06, 0.1, 0.1), frameMat); cassette.position.set(zPos.x, side.h + 0.05, zPos.z); cassette.rotation.y = side.rotY; systemGroup.add(cassette);
          const fabric = new THREE.Mesh(new THREE.BoxGeometry(side.w, side.h, 0.005), zipMat); fabric.rotation.y = side.rotY; fabric.userData = { fullHeight: side.h, topAnchorY: side.h }; fabric.position.set(zPos.x, side.h / 2, zPos.z); systemGroup.add(fabric); zipPanels.push(fabric);
        }
      });

      document.getElementById('glassControls').style.display = hasGlass ? 'block' : 'none';
      document.getElementById('zipControls').style.display = hasZip ? 'block' : 'none';
      updateArea(W, D); updateTechHUD(currentRoof, closedSides, hasInsulatedSystem);
      applyGlassOpenness(); applyZipOpenness(); applyRoofMechanics();
    }

    function applyRoofMechanics() {

      if (currentRoof === 'bioclimatic') {

    function applyGlassOpenness() {
      const ratio = (parseFloat(document.getElementById('glassOpenness').value) || 0) / 100; document.getElementById('glassVal').innerText = ratio <= 0.02 ? 'Kapalı' : (ratio >= 0.98 ? 'Açık' : `%${Math.round(ratio * 100)}`);
      glassPanels.forEach(p => {
        if (p.userData.type === 'hinged_door') p.rotation.y = p.userData.closedAngle + (p.userData.openAngle - p.userData.closedAngle) * ratio;
        else if (p.userData.type === 'guillotine' && !p.userData.isFixed) p.position.y = p.userData.closedY + (p.userData.openY - p.userData.closedY) * ratio;
        else if (p.userData.type === 'sliding' || p.userData.type === 'rts116') { p.position.x = p.userData.closedX + (p.userData.openX - p.userData.closedX) * ratio; p.position.z = p.userData.closedZ + (p.userData.openZ - p.userData.closedZ) * ratio; }
        else if (p.userData.type === 'fd60' || p.userData.type === 'cambalkon') { p.position.x = p.userData.closedX + (p.userData.openX - p.userData.closedX) * ratio; p.position.z = p.userData.closedZ + (p.userData.openZ - p.userData.closedZ) * ratio; p.rotation.y = p.userData.closedRotY + (p.userData.openRotY - p.userData.closedRotY) * ratio; }
      });
    }
