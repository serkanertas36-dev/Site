      const FH = parseFloat(document.getElementById('frontHeight').value);
      const BH = FH + 0.15; 

      houseWall.position.set(0, 5, -D/2 - 0.25);

      function createBox(w, h, d, x, y, z, rx=0, mat=frameMat) {
        const mesh = new THREE.Mesh(new THREE.BoxGeometry(w, h, d), mat);
        mesh.position.set(x, y, z); mesh.rotation.x = rx; mesh.castShadow = true; mesh.receiveShadow = true;
        systemGroup.add(mesh); return mesh;
      }

      createBox(POST_SIZE, FH, POST_SIZE, -W/2 + POST_SIZE/2, FH/2, D/2 - POST_SIZE/2);
      createBox(POST_SIZE, FH, POST_SIZE,  W/2 - POST_SIZE/2, FH/2, D/2 - POST_SIZE/2);
      createBox(POST_SIZE, BH, POST_SIZE, -W/2 + POST_SIZE/2, BH/2, -D/2 + POST_SIZE/2);
      createBox(POST_SIZE, BH, POST_SIZE,  W/2 - POST_SIZE/2, BH/2, -D/2 + POST_SIZE/2);
      createBox(W, BEAM_HEIGHT, BEAM_WIDTH, 0, FH - BEAM_HEIGHT/2, D/2 - BEAM_WIDTH/2);
      createBox(W, BEAM_HEIGHT, BEAM_WIDTH, 0, BH - BEAM_HEIGHT/2, -D/2 + BEAM_WIDTH/2);

      const slopeLength = Math.sqrt(Math.pow(D, 2) + Math.pow(BH - FH, 2));
      const pitchAngle = Math.atan2(BH-FH, D);
      const roofCenterY = (FH + BH)/2 - BEAM_HEIGHT/2 + 0.08;

      createBox(BEAM_WIDTH, BEAM_HEIGHT, slopeLength, -W/2 + BEAM_WIDTH/2, roofCenterY, 0, pitchAngle);
      createBox(BEAM_WIDTH, BEAM_HEIGHT, slopeLength, W/2 - BEAM_WIDTH/2, roofCenterY, 0, pitchAngle);

      if (currentRoof === 'bioclimatic' || currentRoof === 'rolling_roof') {
        const count = Math.ceil(D / 0.22); const step = D / count; 
        for (let i = 0; i < count; i++) {
          const louver = new THREE.Mesh(new THREE.BoxGeometry(W - (BEAM_WIDTH*2), 0.03, step + 0.005), louverMat);
          const baseZ = -D/2 + step * (i + 0.5);
          louver.position.set(0, roofCenterY, baseZ); louver.userData = { baseZ: baseZ, index: i }; 
          louver.castShadow = true; systemGroup.add(louver); roofActionElements.push(louver);
        }
      } else if (currentRoof === 'pergola') {
        const segments = 12; const segStep = D / segments;
        for(let i = 0; i < segments; i++) {
            const fGroup = new THREE.Group();
            const fabric = new THREE.Mesh(new THREE.BoxGeometry(W - 0.15, 0.02, segStep * 0.95), fabricMat);
            fGroup.add(fabric);
            const baseZ = -D/2 + segStep * (i + 0.5); fGroup.position.set(0, roofCenterY, baseZ); fGroup.userData = { baseZ: baseZ, index: i };
            systemGroup.add(fGroup); roofActionElements.push(fGroup);
        }
      } else if (currentRoof === 'veranda') {
        const rG = new THREE.Mesh(new THREE.BoxGeometry(W-0.05, 0.02, slopeLength), glassMat);
        rG.position.set(0, roofCenterY, 0); rG.rotation.x = pitchAngle; systemGroup.add(rG);
        const rafterCount = Math.ceil(W / 0.85); const rafterSpacing = W / rafterCount;
        for(let i=1; i<rafterCount; i++) {
           const rafX = -W/2 + (i * rafterSpacing);
           const rafter = new THREE.Mesh(new THREE.BoxGeometry(0.06, 0.12, slopeLength + 0.1), frameMat);
           rafter.position.set(rafX, roofCenterY - 0.02, 0); rafter.rotation.x = pitchAngle; rafter.castShadow = true;
           systemGroup.add(rafter);
        }
      }

      const sides = [
          { id: 'Front', w: W - (POST_SIZE*2), h: FH - BEAM_HEIGHT, posX: 0, posZ: D/2 - POST_SIZE/2, rotY: 0, xDir: 0, zDir: 1 },
          { id: 'Back', w: W - (POST_SIZE*2), h: BH - BEAM_HEIGHT, posX: 0, posZ: -D/2 + POST_SIZE/2, rotY: 0, xDir: 0, zDir: -1 },
          { id: 'Left', w: D - (POST_SIZE*2), h: (FH+BH)/2 - BEAM_HEIGHT, posX: -W/2 + POST_SIZE/2, posZ: 0, rotY: Math.PI/2, xDir: -1, zDir: 0 },
          { id: 'Right', w: D - (POST_SIZE*2), h: (FH+BH)/2 - BEAM_HEIGHT, posX: W/2 - POST_SIZE/2, posZ: 0, rotY: Math.PI/2, xDir: 1, zDir: 0 }
      ];

      let hasGlass = false; let hasZip = false; let closedSides = 0;
      let hasInsulatedSystem = false;

      sides.forEach(side => {
          const type = document.getElementById('glazing' + side.id).value;
          if(type !== 'none') closedSides++;

          if(type === 'wall') {
              const wall = new THREE.Mesh(new THREE.BoxGeometry(side.w, side.h, 0.08), wallMat);
              wall.position.set(side.posX, side.h/2, side.posZ); wall.rotation.y = side.rotY; wall.castShadow = true; wall.receiveShadow = true; systemGroup.add(wall);
          } else if (type !== 'none') {
              hasGlass = true;
              
              let panelCount = 3;
              if (type === 'fd60') panelCount = 4;
              if (type === 'rts116') panelCount = 2;
              
              if (type === 'rts116' || type === 'rwt64') hasInsulatedSystem = true;

              for(let i=0; i<panelCount; i++) {
                  const pGroup = new THREE.Group();
                  
                  let pw = side.w / panelCount;
                  let ph = side.h;
                  let fThick = 0.04;

                  if (type === 'guillotine') {
                      pw = side.w; ph = side.h / 3;
                  } else if (type === 'sliding') {
                      pw = side.w/3 + 0.05;
                  } else if (type === 'rts116') {
                      pw = side.w/2 + 0.08; fThick = 0.06;
                  } else if (type === 'rwt55') {
                      fThick = 0.055;
                  } else if (type === 'rwt64') {
                      fThick = 0.064;
                  }

                  const framedGlass = createFramedGlass(pw, ph, fThick);
                  pGroup.add(framedGlass); 
                  
                  if (type === 'fd60') {
                      let closedLocalX = -side.w/2 + pw/2 + (i * pw);
                      let closedLocalZ = 0;
                      let closedRotY = side.rotY;

                      let openLocalX = -side.w/2 + (pw * 0.1) + (i * pw * 0.15);
                      let openLocalZ = (i % 2 === 0) ? 0.3 : -0.3;
                      let openRotY = side.rotY + ((i % 2 === 0) ? 1.2 : -1.2);
                      
                      pGroup.userData = { 
                          type: 'fd60', 
                          closedX: side.posX + (side.rotY === 0 ? closedLocalX : closedLocalZ * side.xDir),
                          closedZ: side.posZ + (side.rotY === 0 ? closedLocalZ * side.zDir : closedLocalX),
                          closedRotY,
                          openX: side.posX + (side.rotY === 0 ? openLocalX : openLocalZ * side.xDir),
                          openZ: side.posZ + (side.rotY === 0 ? openLocalZ * side.zDir : openLocalX),
                          openRotY
                      };
                      pGroup.position.set(pGroup.userData.closedX, side.h/2, pGroup.userData.closedZ);
                      pGroup.rotation.y = closedRotY;
                  } else {
                      pGroup.rotation.y = side.rotY;
                      
                      if(type === 'guillotine') {
                          pGroup.userData = { type: 'guillotine', isFixed: (i===0), closedY: ph/2 + (i * ph), openY: ph/2 + (i * 0.05) };
                          let localZ = (1 - i) * 0.04; 
                          let finalX = side.posX + (side.rotY !== 0 ? localZ * side.xDir : 0);
                          let finalZ = side.posZ + (side.rotY === 0 ? localZ * side.zDir : 0);
                          pGroup.position.set(finalX, pGroup.userData.closedY, finalZ);
                      } else if (type === 'sliding' || type === 'rts116') {
                          let overlap = (type === 'rts116') ? 0.08 : 0.05;
                          let effWidth = side.w / panelCount;
                          let localX = -side.w/2 + pw/2 + (i * effWidth) - (i>0 ? overlap/2 : 0); 
                          let openLocalX = -side.w/2 + pw/2 + (i * 0.05);
                          let localZ = (1 - i) * (type === 'rts116' ? 0.06 : 0.04);
                          
                          let closedX = side.posX + (side.rotY === 0 ? localX : localZ * side.xDir);
                          let closedZ = side.posZ + (side.rotY === 0 ? localZ * side.zDir : localX);
                          let openX = side.posX + (side.rotY === 0 ? openLocalX : localZ * side.xDir);
                          let openZ = side.posZ + (side.rotY === 0 ? localZ * side.zDir : openLocalX);
                          
                          pGroup.position.set(closedX, side.h/2, closedZ);
                          pGroup.userData = { type: type, isFixed: (i===0), closedX, closedZ, openX, openZ };
                      } else if (type === 'rwt55' || type === 'rwt64') {
                          let localX = -side.w/2 + pw/2 + (i * pw); 
                          let finalX = side.posX + (side.rotY === 0 ? localX : 0);
                          let finalZ = side.posZ + (side.rotY === 0 ? 0 : localX);
                          pGroup.position.set(finalX, side.h/2, finalZ);
                      }
                  }
                  
                  systemGroup.add(pGroup); 
                  if (type === 'guillotine' || type === 'sliding' || type === 'rts116' || type === 'fd60') {
                      glassPanels.push(pGroup);
                  }
              }
          }

          if (document.getElementById('zip' + side.id).checked) {
              hasZip = true;
              const offsetOut = 0.09; 
              const casX = side.posX + (side.xDir * offsetOut); 
              const casZ = side.posZ + (side.zDir * offsetOut);
              
              const cassette = new THREE.Mesh(new THREE.BoxGeometry(side.w + 0.06, 0.1, 0.1), frameMat);
              cassette.position.set(casX, side.h + 0.05, casZ); 
              cassette.rotation.y = side.rotY; 
              cassette.castShadow = true; 
              systemGroup.add(cassette);
              
              const zipGeo = new THREE.BoxGeometry(side.w, side.h, 0.005); 
              const fabric = new THREE.Mesh(zipGeo, zipMat);
              fabric.rotation.y = side.rotY; 
              fabric.castShadow = true; 
              fabric.userData = { fullHeight: side.h, topAnchorY: side.h };
              fabric.position.set(casX, side.h / 2, casZ); 
              systemGroup.add(fabric); zipPanels.push(fabric);
          }
      });

      document.getElementById('glassControls').style.display = hasGlass ? 'block' : 'none';
      document.getElementById('zipControls').style.display = hasZip ? 'block' : 'none';
      
      updateArea(W, D); updateTechHUD(currentRoof, closedSides, hasInsulatedSystem);
      applyGlassOpenness(); applyZipOpenness(); applyRoofMechanics();
    }

    function updateTechHUD(roof, closedSides, hasInsulatedSystem) {
        const hudSnow = document.getElementById('hud-snow-val');
        const hudWind = document.getElementById('hud-wind-val');
        const hudIns = document.getElementById('hud-insulation');
        const snowBar = document.getElementById('hud-snow-bar');
        const windBar = document.getElementById('hud-wind-bar');

        if(roof === 'bioclimatic') { 
            hudSnow.innerText = '200 kg/m²'; snowBar.style.width = '90%';
            hudWind.innerText = '120 km/h'; windBar.style.width = '95%';
        } else if(roof === 'rolling_roof') { 
            hudSnow.innerText = '150 kg/m²'; snowBar.style.width = '70%';
            hudWind.innerText = '100 km/h'; windBar.style.width = '80%';
        } else if(roof === 'pergola') { 
            hudSnow.innerText = 'N/A'; snowBar.style.width = '0%';
            hudWind.innerText = '80 km/h'; windBar.style.width = '60%';
        } else if(roof === 'veranda') { 
            hudSnow.innerText = '250 kg/m²'; snowBar.style.width = '100%';
            hudWind.innerText = '130 km/h'; windBar.style.width = '100%';
        }

        if(closedSides === 0) hudIns.innerText = 'Low (Open)';
        else if(hasInsulatedSystem) hudIns.innerText = 'Premium Thermal';
        else if(closedSides > 0 && closedSides < 4) hudIns.innerText = 'Medium (Semi-Closed)';
        else hudIns.innerText = 'High (Standard Enclosed)';
    }

    function applyRoofMechanics() {
        const val = parseFloat(document.getElementById('roofActionVal').value); const ratio = val / 100; 
        if (currentRoof === 'bioclimatic') { document.getElementById('roofSliderVal').innerText = `${Math.round(val * 1.1)}°`; roofActionElements.forEach(l => l.rotation.x = (val * 1.1 * Math.PI) / 180); } 
        else if (currentRoof === 'rolling_roof') {
            document.getElementById('roofSliderVal').innerText = `${val}% Retracted`; const D = parseFloat(document.getElementById('depth').value); const stackSpacing = 0.06;
            roofActionElements.forEach((louver, idx) => {
                louver.rotation.x = ratio * (Math.PI / 3); 
                const closedZ = louver.userData.baseZ; const stackedZ = (-D/2 + POST_SIZE/2) + (idx * stackSpacing);
                louver.position.z = closedZ + (stackedZ - closedZ) * ratio;
            });
        }
        else if (currentRoof === 'pergola') {
            document.getElementById('roofSliderVal').innerText = `${val}% Retracted`; const D = parseFloat(document.getElementById('depth').value); const stackSpacing = 0.05;
            roofActionElements.forEach((segment, idx) => {
                const closedZ = segment.userData.baseZ; const stackedZ = (-D/2 + POST_SIZE/2) + (idx * stackSpacing);
                segment.position.z = closedZ + (stackedZ - closedZ) * ratio; segment.scale.z = Math.max(0.2, 1.0 - (ratio * 0.8)); 
            });
        }
    }
    document.getElementById('roofActionVal').addEventListener('input', applyRoofMechanics);

    function applyGlassOpenness() {
        const ratio = parseFloat(document.getElementById('glassOpenness').value) / 100; document.getElementById('glassVal').innerText = ratio === 0 ? 'Closed' : (ratio === 1 ? 'Open' : `${Math.round(ratio*100)}%`);
        glassPanels.forEach((p) => { 
            if(p.userData.type === 'guillotine' && !p.userData.isFixed) { 
                p.position.y = p.userData.closedY + (p.userData.openY - p.userData.closedY) * ratio; 
            } else if ((p.userData.type === 'sliding' || p.userData.type === 'rts116') && !p.userData.isFixed) {
                p.position.x = p.userData.closedX + (p.userData.openX - p.userData.closedX) * ratio;
                p.position.z = p.userData.closedZ + (p.userData.openZ - p.userData.closedZ) * ratio;
            } else if (p.userData.type === 'fd60') {
                p.position.x = p.userData.closedX + (p.userData.openX - p.userData.closedX) * ratio;
                p.position.z = p.userData.closedZ + (p.userData.openZ - p.userData.closedZ) * ratio;
                p.rotation.y = p.userData.closedRotY + (p.userData.openRotY - p.userData.closedRotY) * ratio;
            }
        });
    }
    
    function applyZipOpenness() {
        const ratio = parseFloat(document.getElementById('zipOpenness').value) / 100; 
        document.getElementById('zipVal').innerText = ratio === 0 ? 'Down' : (ratio === 1 ? 'Open' : `${Math.round(ratio*100)}%`);
        zipPanels.forEach((fabric) => { 
            const currentScaleY = Math.max(0.01, 1.0 - ratio);
            fabric.scale.y = currentScaleY;
            fabric.position.y = fabric.userData.topAnchorY - (fabric.userData.fullHeight * currentScaleY) / 2;
        });
    }

    document.getElementById('glassOpenness').addEventListener('input', applyGlassOpenness); document.getElementById('zipOpenness').addEventListener('input', applyZipOpenness);
